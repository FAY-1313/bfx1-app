export type PlayMode = 'guitar' | 'bass';

export interface SubModel {
  id: string;
  /** 数据名 (组合键/预设键/id 依据, 保持稳定避免预设迁移) */
  name: string;
  /** UI 显示名: B&F NAM + 品牌+型号 (较 name 更长, 仅用于展示) */
  display?: string;
  knobs: string[];
  group?: string;
  color?: string;
  accentColor?: string;
  /** 禁用选项: 灰色不可选 (贝斯过载/失真) */
  disabled?: boolean;
  speakerLabel?: string;
  /** 三旋钮布局: 'row'=一排(默认), 'tri'=上2下1 两排 */
  layout?: 'row' | 'tri';
  // 两级结构: captures 为硬件型号下的位置快照列表 (组合键 = "name/pos")
  captures?: { pos: string; label?: string }[];
}

export interface ModuleConfig {
  id: string;
  label: string;
  shortLabel: string;
  hasSubModels: boolean;
  subModels?: { guitar: SubModel[]; bass: SubModel[] };
  knobs?: string[];
  bodyColor?: string;
  accentColor?: string;
}

export function getMergedSubModels(module: ModuleConfig): SubModel[] {
  if (!module.hasSubModels || !module.subModels) return [];
  return [...(module.subModels.guitar ?? []), ...(module.subModels.bass ?? [])];
}

export function getKnobsForModule(module: ModuleConfig, subModelId: string | null, mode: PlayMode): string[] {
  if (!module.hasSubModels) return module.knobs ?? [];
  const models = getSubModels(module, mode);
  const found = models.find((m) => m.id === subModelId);
  return found?.knobs ?? [];
}

// ─── 两级结构辅助: 组合键 = "硬件名/位置id" (与 presetData/parameterMap 组合键一致) ───

export function getSubModelCaptures(sm: SubModel | undefined): { pos: string }[] {
  return sm?.captures ?? [];
}

// 默认组合键: 硬件 + 首个位置快照; 无 captures 的模块返回硬件名本身
export function getDefaultSubModelName(sm: SubModel): string {
  if (sm.captures?.length) return `${sm.name}/${sm.captures[0].pos}`;
  return sm.name;
}

// 拆分组合键 "硬件/位置" -> { hardware, pos? }
export function splitSubModelName(composed: string): { hardware: string; pos?: string } {
  const i = composed.lastIndexOf('/');
  if (i <= 0) return { hardware: composed };
  return { hardware: composed.slice(0, i), pos: composed.slice(i + 1) };
}
