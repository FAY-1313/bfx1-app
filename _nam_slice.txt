function NamPanel({
  visible,
  onClose,
  namFile,
  onNameFileChange,
  irFile,
  onIrFileChange,
}: {
  visible: boolean;
  onClose: () => void;
  namFile: string | null;
  onNameFileChange: (name: string) => void;
  irFile: string | null;
  onIrFileChange: (f: string | null) => void;
}) {
  const { t } = useI18n();
  const [activeMode, setActiveMode] = useState<'nam' | 'ir'>('nam');
  const [t3kTones, setT3kTones] = useState<api.Tone3000Tone[]>([]);
  const [t3kLoading, setT3kLoading] = useState(false);
  const [t3kPlatformFilter, setT3kPlatformFilter] = useState<string>('all');
  const [t3kGearFilter, setT3kGearFilter] = useState<string>('all');
  const [t3kArchFilter, setT3kArchFilter] = useState<string>('all');
  const [t3kPage, setT3kPage] = useState(1);
  const [namResult, setNamResult] = useState<NAMIRResult | null>(null);
  const [searchQuery, setSearchQuery] = useState('');
  const [fullRegs, setFullRegs] = useState<{regAddr:number;value:number;label:string}[]>([]);
  const [applying, setApplying] = useState(false);

  // Load TONE3000 tones when panel becomes visible
  useEffect(() => {
    if (visible && t3kTones.length === 0 && !t3kLoading) {
      loadTones();
    }
  }, [visible]);

  const loadTones = async () => {
    setT3kLoading(true);
    try {
      const data = await api.getTone3000Tones({
        page: 1, limit: 50,
        platform: t3kPlatformFilter !== 'all' ? t3kPlatformFilter : undefined,
        gear_type: t3kGearFilter !== 'all' ? t3kGearFilter : undefined,
        search: searchQuery || undefined,
      });
      setT3kTones(data.tones || []);
    } catch (e: any) {
      console.error("TONE3000 load error:", e);
      toast.error(t('toast.tone3000Fail'));
    } finally {
      setT3kLoading(false);
    }
  };

  const handleFileLoad = () => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = '.nam,.aida-x,.wav,.aiff,.flac';
    input.onchange = (e: any) => {
      const file = e.target?.files?.[0] as File | undefined;
      if (!file) return;
      onNameFileChange(file.name);
      const result = identifyFile(file.name, file.size);
      setNamResult(result);
      if (result.confidence && result.confidence >= 0.3) {
        toast.success(`${t('toast.recognized')}${result.description} (${Math.round(result.confidence * 100)}%)`);
      } else {
        toast.info(t('toast.fileLoaded') + file.name);
      }
    };
    input.click();
  };

  const handleIrFileLoad = () => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = '.ir,.wav,.aiff,.flac';
    input.onchange = (e: any) => {
      const file = e.target?.files?.[0] as File | undefined;
      if (!file) return;
      onIrFileChange(file.name);
      const result = identifyFile(file.name, file.size);
      setNamResult(result);
      if (result.confidence && result.confidence >= 0.3) {
        toast.success(`${t('toast.recognized')}${result.description} (${Math.round(result.confidence * 100)}%)`);
      } else {
        toast.info(t('toast.irLoaded') + file.name);
      }
    };
    input.click();
  };

  const handleT3kLoadTone = async (tone: api.Tone3000Tone) => {
    const ext = tone.platform === 'ir' ? 'wav' : 'nam';
    const filename = `${tone.name.replace(/\s+/g, '_')}.${ext}`;
    if (activeMode === 'ir') onIrFileChange(filename);
    else onNameFileChange(filename);
    const result = identifyFile(filename, 500000);
    setNamResult(result);
    toast.success(t('toast.loaded') + tone.name);
  };

  const handleApplyToHardware = () => {
    if (!namResult) return;
    const regs = namResult.registerSuggestions.map(s => ({
      regAddr: s.regAddr,
      value: s.value,
      label: `Reg ${s.regAddr}`,
    }));
    writeRegs(regs);
  };

  const handleApplyFullNAM = async () => {
    if (!namResult) return;
    setApplying(true);
    const ampType = namResult.description.toLowerCase();
    const fullRegsList = generateNAMConfig(namResult.confidence ?? 0.5, ampType);
    setFullRegs(fullRegsList);
    toast.info(t('toast.generatedNam', { count: fullRegsList.length }));
    await writeRegs(fullRegsList);
    setApplying(false);
  };

  const handleApplyIRCabinet = async () => {
    if (!irFile) { toast.error(t('toast.loadIrFirst')); return; }
    setApplying(true);
    try {
      // Generate a target frequency curve based on IR file type detection
      const confidence = namResult?.confidence ?? 0.5;
      const mags = new Float32Array(35);
      // Generate natural guitar cab curves based on IR characteristics
      const cabType = irFile.toLowerCase();
      for (let i = 0; i < 35; i++) {
        const freq = ISO_35_BAND_FREQS[i] || (20 * Math.pow(1.26, i));
        // Natural cab: slight low-mid bump, highs roll off above 5kHz
        let target = 0.5;
        if (freq < 100) target = 0.3 + (freq / 100) * 0.3;                     // sub-bass rolloff
        else if (freq < 600) target = 0.6 + Math.sin((freq - 100) / 500 * Math.PI) * 0.25; // low-mid bump
        else if (freq < 2000) target = 0.5 - (freq - 600) / 1400 * 0.15;       // slight mids cut
        else if (freq < 5000) target = 0.35 - (freq - 2000) / 3000 * 0.1;      // presence dip
        else target = Math.max(0.15, 0.25 - (freq - 5000) / 15000 * 0.15);     // highs rolloff

        // Add IR-specific flavor
        if (cabType.includes('vintage') || cabType.includes('old')) target *= 0.85;
        if (cabType.includes('modern') || cabType.includes('v30')) target = Math.min(1, target * 1.15);
        if (cabType.includes('bass')) target = Math.min(1, target * 1.3);

        mags[i] = Math.max(0.05, Math.min(1, target * (0.8 + confidence * 0.4)));
      }
      const config = generateIRConfig(mags);
      const regs = irConfigToRegisters(config);
      setFullRegs(regs);
      toast.info(t('toast.generatedPeq', { count: regs.length }));
      await writeRegs(regs);
    } catch (e) {
      console.error('IR apply error:', e);
      toast.error(t('toast.applyIrFail'));
    }
    setApplying(false);
  };

  async function writeRegs(regs: {regAddr:number;value:number;label:string}[]) {
    let count = 0;
    for (const r of regs) {
      const ok = await bleManager.writeRegister(r.regAddr, r.value).catch(() => false);
      if (ok) count++;
    }
    toast.success(t('toast.regsWritten', { count, total: regs.length }));
  }

  const allFiltered = t3kTones.filter(t => {
    if (t3kArchFilter !== 'all' && t.architect !== t3kArchFilter) return false;
    return true;
  });
  const filteredTones = allFiltered.slice(0, t3kPage * 4);

  return (
    <div style={{
      position: 'absolute', bottom: 0, left: 0, right: 0, height: 500,
      background: 'rgba(4,7,12,0.98)',
      borderTop: `1px solid ${visible ? 'rgba(240,111,33,0.25)' : 'transparent'}`,
      opacity: visible ? 1 : 0,
      pointerEvents: visible ? 'auto' : 'none',
      // B4: 常驻面板靠 visible 过渡上滑淡入 (root 自身 transform; 打开稳态为 none, 不影响内部 fixed 元素)
      transform: visible ? 'none' : 'translateY(6px)',
      transition: 'opacity 0.18s, transform 0.18s',
      display: 'flex', flexDirection: 'column',
      zIndex: 60,
    }}>
      {/* Header */}
      <PanelHeader
        title={<>Neural Amp Modeler<span style={{ padding: '1px 5px', borderRadius: 4, background: 'rgba(240,111,33,0.1)', border: '1px solid rgba(240,111,33,0.25)' }}><span style={{ fontSize: 8, color: NAM_ACCENT, letterSpacing: '0.14em', fontFamily: '"Barlow-Regular_2",sans-serif', fontWeight: 700 }}>TONE3000</span></span></>}
        accent={NAM_ACCENT}
        onClose={onClose}
      />

      {/* File loader tabs + rows */}
      <div style={{ padding: '8px 14px 10px', borderBottom: '1px solid rgba(240,111,33,0.08)', flexShrink: 0 }}>
        <div style={{ display: 'flex', gap: 4, marginBottom: 8 }}>
          {(['nam', 'ir'] as const).map(mode => {
            const isActive = activeMode === mode;
            const color = mode === 'nam' ? NAM_ACCENT : '#4cc9f0';
            return (
              <button key={mode} onClick={() => { setActiveMode(mode); setNamResult(null); }}
                style={{
                  flex: 1, height: 32, borderRadius: 4, cursor: 'pointer', fontFamily: '"Barlow-Regular_2",sans-serif',
                  fontSize: 9, letterSpacing: '0.18em', fontWeight: 700, textTransform: 'uppercase',
                  background: isActive ? `${color}18` : 'rgba(255,255,255,0.03)',
                  border: `1px solid ${isActive ? color : 'rgba(255,255,255,0.08)'}`,
                  color: isActive ? color : 'rgba(255,255,255,0.3)',
                  boxShadow: isActive ? `0 0 10px ${color}22` : 'none',
                  transition: 'all 0.15s',
                }}
              >{mode === 'nam' ? '.NAM Model' : '.IR Cabinet'}</button>
            );
          })}
        </div>
        {activeMode === 'nam' && (
          <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
            <button onClick={handleFileLoad}
              style={{ display: 'flex', alignItems: 'center', gap: 5, padding: '5px 10px', borderRadius: 4, cursor: 'pointer', background: 'rgba(240,111,33,0.1)', border: `1px solid rgba(240,111,33,0.3)`, color: NAM_ACCENT, fontFamily: '"Barlow-Regular_2",sans-serif', fontSize: 9, letterSpacing: '0.12em', fontWeight: 700, textTransform: 'uppercase', boxShadow: '0 0 8px rgba(240,111,33,0.1)', flexShrink: 0 }}
            ><FileAudioIcon size={12} color={NAM_ACCENT} />Load .NAM</button>
            <div style={{ flex: 1, padding: '4px 8px', borderRadius: 3, background: 'rgba(240,111,33,0.04)', border: '1px solid rgba(240,111,33,0.12)', overflow: 'hidden' }}>
              <span style={{ fontSize: 9, color: namFile ? 'rgba(255,180,140,0.8)' : 'rgba(240,111,33,0.3)', fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.08em', whiteSpace: 'nowrap', textOverflow: 'ellipsis', display: 'block', overflow: 'hidden' }}>{namFile ?? 'No file loaded'}</span>
            </div>
          </div>
        )}
        {activeMode === 'ir' && (
          <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
            <button onClick={handleIrFileLoad}
              style={{ display: 'flex', alignItems: 'center', gap: 5, padding: '5px 10px', borderRadius: 4, cursor: 'pointer', background: 'rgba(76,201,240,0.08)', border: `1px solid rgba(76,201,240,0.28)`, color: '#4cc9f0', fontFamily: '"Barlow-Regular_2",sans-serif', fontSize: 9, letterSpacing: '0.12em', fontWeight: 700, textTransform: 'uppercase', boxShadow: '0 0 8px rgba(76,201,240,0.08)', flexShrink: 0 }}
            ><FileAudioIcon size={12} color="#4cc9f0" />Load .IR</button>
            <div style={{ flex: 1, padding: '4px 8px', borderRadius: 3, background: 'rgba(76,201,240,0.03)', border: '1px solid rgba(76,201,240,0.12)', overflow: 'hidden' }}>
              <span style={{ fontSize: 9, color: irFile ? 'rgba(140,230,255,0.85)' : 'rgba(76,201,240,0.28)', fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.08em', whiteSpace: 'nowrap', textOverflow: 'ellipsis', display: 'block', overflow: 'hidden' }}>{irFile ?? 'No IR loaded'}</span>
            </div>
          </div>
        )}
        {/* NAM identification result */}
        {namResult && (
          <div style={{ marginTop: 6, padding: '6px 8px', borderRadius: 4, background: 'rgba(240,111,33,0.06)', border: '1px solid rgba(240,111,33,0.15)' }}>
            <div style={{ fontSize: 8, color: 'rgba(255,180,140,0.8)', fontFamily: '"Barlow-Regular_2",sans-serif', fontWeight: 600, letterSpacing: '0.06em' }}>{namResult.description}</div>
            <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', marginTop: 2 }}>
              <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
                <span style={{ fontSize: 8, color: namResult.confidence && namResult.confidence > 0.5 ? '#3ddc84' : 'rgba(255,180,140,0.5)', fontFamily: '"Barlow-Regular_2",sans-serif' }}>
                  {namResult.confidence ? `${Math.round(namResult.confidence * 100)}% match` : 'Unknown'}
                </span>
                <span style={{ fontSize: 8, color: 'rgba(255,180,140,0.4)', fontFamily: '"Barlow-Regular_2",sans-serif' }}>
                  {namResult.registerSuggestions.length} regs
                </span>
              </div>
              {namResult.registerSuggestions.length > 0 && (
                <button onClick={handleApplyToHardware}
                  style={{ padding: '3px 7px', borderRadius: 3, background: 'rgba(240,111,33,0.15)', border: '1px solid rgba(240,111,33,0.35)', color: NAM_ACCENT, fontSize: 8, fontFamily: '"Barlow-Regular_2",sans-serif', fontWeight: 700, cursor: 'pointer', letterSpacing: '0.08em' }}>
                  Apply Basic
                </button>
              )}
            </div>
            {/* Full NAM / IR Cabinet buttons */}
            <div style={{ display: 'flex', gap: 4, marginTop: 5 }}>
              {activeMode === 'nam' && (
                <button onClick={handleApplyFullNAM} disabled={applying}
                  style={{ flex: 1, padding: '4px 6px', borderRadius: 3, cursor: applying?'wait':'pointer', background: 'rgba(240,111,33,0.12)', border: '1px solid rgba(240,111,33,0.3)', color: NAM_ACCENT, fontSize: 8, fontWeight: 700, fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.06em', opacity: applying?0.6:1 }}>
                  {applying ? 'Applying...' : 'Apply Full NAM (50+ regs)'}
                </button>
              )}
              {activeMode === 'ir' && (
                <button onClick={handleApplyIRCabinet} disabled={applying || !irFile}
                  style={{ flex: 1, padding: '4px 6px', borderRadius: 3, cursor: applying?'wait':'pointer', background: 'rgba(76,201,240,0.1)', border: '1px solid rgba(76,201,240,0.3)', color: '#4cc9f0', fontSize: 8, fontWeight: 700, fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.06em', opacity: (applying||!irFile)?0.6:1 }}>
                  {applying ? 'Applying...' : `Apply IR Cab (35-band PEQ)`}
                </button>
              )}
            </div>
            {/* Show generated register count */}
            {fullRegs.length > 0 && (
              <div style={{ marginTop: 4, fontSize: 8, color: 'rgba(255,255,255,0.3)', fontFamily: '"Barlow-Regular_2",sans-serif' }}>
                {fullRegs.length} registers generated — tap to write via BLE
              </div>
            )}
          </div>
        )}
      </div>

      {/* Community section */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 6, padding: '8px 14px 5px', flexShrink: 0 }}>
        <UsersIcon size={10} color="rgba(255,130,80,0.6)" />
        <span style={{ fontSize: 9, color: 'rgba(255,130,80,0.6)', letterSpacing: '0.18em', fontFamily: '"Barlow-Regular_2",sans-serif', fontWeight: 700, textTransform: 'uppercase' }}>{t('preset.communityTones')}</span>
        <div style={{ flex: 1, height: 1, background: 'linear-gradient(90deg,rgba(240,111,33,0.15),transparent)' }} />
        {t3kLoading && <span style={{ fontSize: 8, color: 'rgba(255,130,80,0.4)' }}>{t('preset.loading')}</span>}
      </div>

      <div className="no-scrollbar" style={{ flex: 1, overflowY: 'auto', padding: '0 10px 12px' }}>
        {/* Search bar */}
        <div style={{ display: 'flex', gap: 4, marginBottom: 8 }}>
          <div style={{ flex: 1, display: 'flex', alignItems: 'center', background: 'rgba(240,111,33,0.05)', border: '1px solid rgba(240,111,33,0.15)', borderRadius: 4, padding: '0 8px' }}>
            <SearchIcon size={12} color="rgba(240,111,33,0.4)" />
            <input
              value={searchQuery}
              onChange={e => setSearchQuery(e.target.value)}
              onKeyDown={e => { if (e.key === 'Enter') loadTones(); }}
              placeholder="Search tones..."
              style={{ flex: 1, background: 'none', border: 'none', outline: 'none', color: 'rgba(255,255,255,0.7)', fontSize: 10, padding: '6px 6px', fontFamily: '"Barlow-Regular_2",sans-serif' }}
            />
            <button onClick={loadTones} style={{ background: 'none', border: 'none', cursor: 'pointer', color: 'rgba(240,111,33,0.5)', fontSize: 9, padding: 0 }}>Go</button>
          </div>
        </div>
        {/* Filters */}
        <div style={{ display: 'flex', gap: 4, marginBottom: 8, flexWrap: 'wrap' }}>
          <select value={t3kPlatformFilter} onChange={e => { setT3kPlatformFilter(e.target.value); setT3kTones([]); }}
            style={{ flex: 1, minWidth: 70, height: 24, borderRadius: 3, background: 'rgba(240,111,33,0.07)', border: '1px solid rgba(240,111,33,0.2)', color: 'rgba(255,180,140,0.8)', fontSize: 8, fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.08em', outline: 'none', cursor: 'pointer' }}>
            <option value="all">All Platforms</option>
            <option value="nam">NAM</option>
            <option value="ir">IR</option>
            <option value="aida-x">AIDA-X</option>
            <option value="aa-snapshot">AA Snapshot</option>
            <option value="proteus">Proteus</option>
          </select>
          <select value={t3kGearFilter} onChange={e => { setT3kGearFilter(e.target.value); setT3kTones([]); }}
            style={{ flex: 1, minWidth: 70, height: 24, borderRadius: 3, background: 'rgba(240,111,33,0.07)', border: '1px solid rgba(240,111,33,0.2)', color: 'rgba(255,180,140,0.8)', fontSize: 8, fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.08em', outline: 'none', cursor: 'pointer' }}>
            <option value="all">All Gear</option>
            <option value="amp">Amp</option>
            <option value="full-rig">Full Rig</option>
            <option value="pedal">Pedal</option>
            <option value="outboard">Outboard</option>
            <option value="ir">IR</option>
          </select>
          <select value={t3kArchFilter} onChange={e => setT3kArchFilter(e.target.value)}
            style={{ flex: 1, minWidth: 60, height: 24, borderRadius: 3, background: 'rgba(240,111,33,0.07)', border: '1px solid rgba(240,111,33,0.2)', color: 'rgba(255,180,140,0.8)', fontSize: 8, fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.08em', outline: 'none', cursor: 'pointer' }}>
            <option value="all">All Arch</option>
            <option value="A1">A1</option>
            <option value="A2">A2</option>
          </select>
        </div>
        {/* Tone cards */}
        {filteredTones.length === 0 && t3kLoading && (
          <div style={{ textAlign: 'center', padding: 20, color: 'rgba(255,130,80,0.4)', fontSize: 9, fontFamily: '"Barlow-Regular_2",sans-serif' }}>Loading tones...</div>
        )}
        {filteredTones.length === 0 && !t3kLoading && t3kTones.length === 0 && (
          <div style={{ textAlign: 'center', padding: 20, color: 'rgba(255,130,80,0.4)', fontSize: 9, fontFamily: '"Barlow-Regular_2",sans-serif' }}>No tones found. Try different filters or tap search.</div>
        )}
        {filteredTones.map(tone => (
          <div key={tone.id} style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '7px 8px', borderRadius: 5, marginBottom: 4, background: 'rgba(240,111,33,0.04)', border: '1px solid rgba(240,111,33,0.1)' }}>
            <div style={{ flex: 1, minWidth: 0 }}>
              <div style={{ fontSize: 10, fontWeight: 700, color: 'rgba(255,180,140,0.9)', fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.06em', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{tone.name}</div>
              <div style={{ display: 'flex', alignItems: 'center', gap: 5, marginTop: 2 }}>
                <span style={{ fontSize: 8, color: 'rgba(255,130,80,0.5)', fontFamily: '"Barlow-Regular_2",sans-serif' }}>@{tone.author || 'unknown'}</span>
                <div style={{ padding: '0 4px', borderRadius: 2, background: 'rgba(240,111,33,0.12)', border: '1px solid rgba(240,111,33,0.25)' }}>
                  <span style={{ fontSize: 8, color: NAM_ACCENT, fontFamily: '"Barlow-Regular_2",sans-serif', letterSpacing: '0.1em', fontWeight: 700 }}>{(tone.gear_type || tone.platform).toUpperCase()}</span>
                </div>
                <div style={{ display: 'flex', alignItems: 'center', gap: 2 }}>
                  <DownloadIcon size={8} color="rgba(255,130,80,0.4)" />
                  <span style={{ fontSize: 8, color: 'rgba(255,130,80,0.4)', fontFamily: '"Barlow-Regular_2",sans-serif' }}>{(tone.downloads || 0).toLocaleString()}</span>
                </div>
              </div>
            </div>
            <button onClick={() => handleT3kLoadTone(tone)}
              style={{ padding: '4px 10px', borderRadius: 3, background: 'rgba(240,111,33,0.12)', border: `1px solid rgba(240,111,33,0.3)`, color: NAM_ACCENT, fontSize: 9, fontWeight: 700, letterSpacing: '0.12em', fontFamily: '"Barlow-Regular_2",sans-serif', cursor: 'pointer', textTransform: 'uppercase', flexShrink: 0 }}>
              Load
            </button>
          </div>
        ))}
        {filteredTones.length < allFiltered.length && (
          <button onClick={() => setT3kPage(p => p + 1)}
            style={{ width: '100%', marginTop: 4, padding: '6px 0', borderRadius: 4, background: 'rgba(240,111,33,0.06)', border: '1px solid rgba(240,111,33,0.15)', color: 'rgba(240,111,33,0.5)', fontSize: 9, fontWeight: 700, letterSpacing: '0.12em', fontFamily: '"Barlow-Regular_2",sans-serif', cursor: 'pointer', textTransform: 'uppercase' }}>
            Load More
          </button>
        )}
      </div>
    </div>
  );
}
